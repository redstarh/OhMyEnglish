#!/usr/bin/env python3
"""`TASK-119` AC#1 — 모델이 `level` 객체에 붙이는 여분·기형 키의 재현률을 센다.

⛔ DB 는 SELECT 만 한다. `build_plan_prompt` → Claude → `parse_plan` 까지만 하고 계획 행·노트
행을 쓰지 않는다.

⛔ 프롬프트를 여기서 바꾸지 않는다 — 재는 것은 **지금 제품이 보내는 것**이다(커밋 `bee48cc` 판).
문면 수정 판은 2단계이고 이 회차의 상한 안에서 하지 않는다.

거부 사유를 세 갈래로 가른다: JSON 문법 위반 · `extra_forbidden`(여분 키) · 그 외.
⛔ 뭉개면 「그 외」가 이 부류로 잘못 셈된다.
"""

import asyncio
import json
import re
from pathlib import Path

from app.config import get_settings
from app.db import close_pool, pool
from app.models.plan import parse_plan
from app.services.chronic import deepest_recurrence
from app.services.plan import build_plan_prompt
from app.services.plan_input import load_plan_input
from app.workers.claude_client import BedrockClaudeClient

ROUNDS = 5  # 회차 기록 §0 의 1단계 상한. ⛔ 여기서 늘리지 않는다.
_OUT = Path("/tmp/task119_counts.json")
_EXTRA_FIELD = re.compile(r"^(\S+)\n\s+Extra inputs are not permitted", re.MULTILINE)


def classify(raw: str, verdict: str) -> tuple[str, str]:
    """(갈래, 세부) — 갈래는 통과 · 여분키 · JSON문법 · 그외 넷이다."""
    if verdict == "통과":
        return "통과", ""
    if "Extra inputs are not permitted" in verdict:
        found = _EXTRA_FIELD.search(verdict)
        return "여분키", found.group(1) if found else "(경로 미확인)"
    try:
        json.loads(raw)
    except json.JSONDecodeError as exc:
        return "JSON문법", str(exc)
    return "그외", verdict.splitlines()[0]


async def main() -> None:
    settings = get_settings()
    conn_pool = await pool()

    async with conn_pool.acquire() as conn:
        user_id = await conn.fetchval(
            "select ls.user_id from session_plans sp "
            "join learning_sessions ls on ls.id = sp.session_id "
            "order by sp.created_at desc limit 1"
        )
        if user_id is None:
            raise SystemExit("계획이 붙은 세션이 없다 — 이 팔은 제품 재료 위에 선다")
        data = await load_plan_input(conn, user_id)

    deepest = deepest_recurrence(data.chronic)
    prompt = build_plan_prompt(data)
    allowed = {r.pattern_id for r in data.due_reviews} | {m.pattern_id for m in data.chronic}
    print(f"=== 재료 user {user_id} · 만성 {len(data.chronic)}건 · 프롬프트 {len(prompt)}자")

    client = BedrockClaudeClient(settings)
    rounds: list[dict[str, str]] = []
    for number in range(1, ROUNDS + 1):
        raw = await client.analyze(prompt)
        Path(f"/tmp/task119_raw_{number}.json").write_text(raw, encoding="utf-8")
        verdict = "통과"
        try:
            parse_plan(
                raw,
                current_level=data.current_level,
                allowed_pattern_ids=allowed,
                deepest_pattern_id=None if deepest is None else deepest.pattern_id,
            )
        except Exception as exc:  # noqa: BLE001 — 사유를 회차 기록에 그대로 옮긴다
            verdict = f"{type(exc).__name__}: {exc}"
        kind, detail = classify(raw, verdict)
        print(f"  {number}/{ROUNDS} {kind} {detail}"[:200])
        rounds.append({"kind": kind, "detail": detail, "verdict": verdict})

    tally: dict[str, int] = {}
    for row in rounds:
        tally[row["kind"]] = tally.get(row["kind"], 0) + 1
    print(f"=== 집계 {tally}")
    _OUT.write_text(
        json.dumps(
            {"user_id": str(user_id), "prompt_chars": len(prompt), "rounds": rounds, "tally": tally},
            ensure_ascii=False,
            indent=2,
        ),
        encoding="utf-8",
    )
    print(f"→ {_OUT}")
    await close_pool()


if __name__ == "__main__":
    asyncio.run(main())
