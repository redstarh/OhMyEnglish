#!/usr/bin/env python3
"""TASK-128.4 AC#1 — 「발음 집중」으로 들어갔을 때 화면이 무엇을 말하는지 관측한다.

⛔ **판정하지 않는다** — 문구를 인쇄하고 화면을 저장한다(판정은 회차 기록이 한다).
⛔ **Nova 를 쓰지 않는다** — `VOICE_ADAPTER=stub` 으로 충분하다. 재는 것은 `session_started` 가
실어 온 `pronunciation_focus` 의 «부재»에 화면이 붙이는 문구이고, 그 값은 `api/ws.py` 가 어댑터와
**무관하게** 계산한다.
⚠️ 마이크는 전용 Chrome 의 가짜 장치로 연다(함정 `H-BD`). 합성 클릭은 user activation 을 만들지
못하므로 `Cdp.click`(실제 입력)을 쓴다(함정 `H-AE`).

실행:
    cd app/backend && .venv/bin/python \
      ../../tests/harness/runs/2026-09-15-task116-2-review-excluded-card/enter.py.txt \
      --port 9333 --base http://localhost:3001 --out <경로>.png
"""

from __future__ import annotations

import argparse
import asyncio
import base64
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))

from c2_render_hierarchy import Cdp, find_target  # noqa: E402
from websockets.asyncio.client import connect  # noqa: E402

_LINES_JS = """
(() => JSON.stringify({
  url: location.href,
  paragraphs: Array.from(document.querySelectorAll('p')).map((p) => (p.textContent || '').trim())
    .filter((t) => t.length > 0),
  buttons: Array.from(document.querySelectorAll('button')).map((b) => (b.textContent || '').trim()),
}))()
"""

# ⚠️ 진입 문구는 **세션이 살아 있는 동안만** 화면에 있다 — stub 어댑터에서는 세션이 곧 끝나
# 결과 화면으로 넘어가므로 **폴링으로는 놓친다**(실측: 100ms 폴링 · 1.2초 · 15초 모두 놓쳤고
# 결과 화면만 보였다). ⇒ 클릭 **전**에 `MutationObserver` 를 걸어 «한 번이라도 그려진» 문장을
# 전부 모은다. 그러면 20ms 만 살아 있던 페인트도 잡힌다.
_INSTALL_WATCH_JS = """
(() => {
  window.__seenTexts = [];
  const record = (root) => {
    if (!root || !root.querySelectorAll) return;
    for (const p of root.querySelectorAll('p')) {
      const t = (p.textContent || '').trim();
      if (t && !window.__seenTexts.includes(t)) window.__seenTexts.push(t);
    }
  };
  record(document.body);
  const observer = new MutationObserver((records) => {
    for (const r of records) {
      record(r.target);
      for (const node of r.addedNodes) record(node);
    }
  });
  observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  window.__seenObserver = observer;
  return 'watch-installed';
})()
"""

_READ_WATCH_JS = """
JSON.stringify({
  notice: (window.__seenTexts || []).find((t) => /발음 연습으로 시작했어요/.test(t)) || null,
  seen: window.__seenTexts || [],
  url: location.href,
})
"""


async def main_async(args: argparse.Namespace) -> int:
    home = f"{args.base}/"
    ws_url, _ = find_target(args.port, home)
    async with connect(ws_url, max_size=None) as ws:
        cdp = Cdp(ws)
        await cdp.call("Page.enable")
        await cdp.call("Page.navigate", {"url": home})
        await asyncio.sleep(3)
        installed = await cdp.eval(_INSTALL_WATCH_JS)
        if installed != "watch-installed":
            raise SystemExit(f"관측기 설치 반환값이 예상과 다르다: {installed!r}")
        await cdp.click(
            "Array.from(document.querySelectorAll('button'))"
            "  .find(b => /발음 집중/.test(b.textContent||''))",
            "발음 집중",
        )
        await asyncio.sleep(args.settle_ms / 1000)
        print(f"관측: {await cdp.eval(_READ_WATCH_JS)}")
        print(f"지금 화면: {await cdp.eval(_LINES_JS)}")
        shot = await cdp.call("Page.captureScreenshot", {"format": "png"})
        Path(args.out).write_bytes(base64.b64decode(shot["data"]))
        print(f"스크린샷: {args.out}")
    return 0


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--port", type=int, default=9333)
    ap.add_argument("--base", default="http://localhost:3001")
    ap.add_argument("--out", required=True)
    ap.add_argument("--settle-ms", type=int, default=12000)
    return asyncio.run(main_async(ap.parse_args()))


if __name__ == "__main__":
    sys.exit(main())
