#!/usr/bin/env python3
"""`TASK-103` AC#2 — 결과 화면을 열어 보기 위한 씨앗.

⛔ 검증 전용 DB 에만 쓴다(`DATABASE_URL` 로 받는다). 공유 dev DB 를 건드리지 않는다.
⛔ 화면이 「시범 문장」 자리에 무엇을 렌더하는지 보려는 것이므로 **실제 생명주기를 그대로 탄다** —
   pending(무너진 전사) → 판정(옳은 목표 문장). SQL 로 직접 넣으면 고친 UPDATE 경로를 안 지난다.
"""

from __future__ import annotations

import asyncio
import os
import sys

import asyncpg

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", "..", "..", "..", "app", "backend"))

from app.services.pronunciation import record_attempt  # noqa: E402

BROKEN = "I think Sri sings are ready for the demo."
MODELED = "I think three things are ready for the demo."
HEARD = "i think sri, sings are ready for the demo."


async def main() -> None:
    dsn = os.environ["DATABASE_URL"]
    conn = await asyncpg.connect(dsn)
    try:
        user_id = await conn.fetchval(
            "insert into users (display_name, timezone, current_level) "
            "values ('TASK-103 screen check', 'Asia/Seoul', 'A2') returning id"
        )
        session_id = await conn.fetchval(
            "insert into learning_sessions (user_id, mode, ended_at) "
            "values ($1, 'speaking', now()) returning id",
            user_id,
        )
        await conn.execute(
            "insert into utterances (session_id, speaker, transcript, sequence_no) "
            "values ($1, 'user', $2, 1)",
            session_id,
            HEARD,
        )
        await record_attempt(conn, session_id, target_form=BROKEN, outcome="pending")
        await record_attempt(
            conn,
            session_id,
            target_form=MODELED,
            outcome="correct",
            spoken_form=HEARD,
            target_sound="th_as_s",
        )
        rows = await conn.fetch(
            "select target_form, outcome from pronunciation_attempts where session_id = $1",
            session_id,
        )
        print(f"session_id = {session_id}")
        for row in rows:
            print(f"  저장된 target_form = {row['target_form']!r} · outcome = {row['outcome']}")
    finally:
        await conn.close()


if __name__ == "__main__":
    asyncio.run(main())
