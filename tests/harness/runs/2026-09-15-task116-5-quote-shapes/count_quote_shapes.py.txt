#!/usr/bin/env python3
"""TASK-116.5 AC#1 — 코치 발화의 «인용 모양»을 센다. ⛔ 새 회차를 돌리지 않는다.

재료는 이미 있는 프레임 증거다(`.harness/evidence/*-frames.json` — 회차마다 하네스가 남긴 원본).
그 파일들의 `speaker == "agent"` final 을 모아, 각 발화가 어떤 모양으로 인용했는지 센다.

분류는 **제품 코드의 문턱을 그대로 쓴다** — `pronunciation._QUOTED_TOKEN_RE` 를 import 한다.
⛔ 정규식을 여기 다시 쓰지 않는다(두 곳에 두면 한쪽이 낡는다).

| 모양 | 뜻 |
|---|---|
| `sound` | 문턱에 걸리는 1~3자 토큰이 있다 — 검사가 갈래 ①로 판정한다 |
| `word` | 낱말/하이픈 토큰만 있다 — 갈래 ②로 판정한다(결정 98) |
| `sentence_only` | **따옴표는 있는데 토큰이 0개다** — 문장 전체 인용이고 검사가 판정하지 못한다 |
| `none` | 인용이 없다 |

실행:
    cd app/backend && .venv/bin/python \
      ../../tests/harness/runs/2026-09-15-task116-5-quote-shapes/count_quote_shapes.py.txt
"""

from __future__ import annotations

import json
import pathlib
import re
from collections import Counter

from app.services.pronunciation import _QUOTED_TOKEN_RE, _SOUND_MAX_LEN

EVIDENCE = pathlib.Path(__file__).resolve().parents[4] / ".harness" / "evidence"
_QUOTE_CHARS = re.compile(r"['\"‘’“”]")


def shape(text: str) -> str:
    tokens = [m.group(1) for m in _QUOTED_TOKEN_RE.finditer(text)]
    if not tokens:
        return "sentence_only" if _QUOTE_CHARS.search(text) else "none"
    if any(len(t) <= _SOUND_MAX_LEN and "-" not in t for t in tokens):
        return "sound"
    return "word"


def agent_finals(payload: object) -> list[str]:
    """프레임 파일의 모양이 회차마다 다르다 — 리스트이거나 `frames` 키를 갖거나 중첩이다."""
    out: list[str] = []
    stack = [payload]
    while stack:
        node = stack.pop()
        if isinstance(node, dict):
            if node.get("type") == "final" and node.get("speaker") == "agent":
                text = node.get("text")
                if isinstance(text, str) and text.strip():
                    out.append(text)
            stack.extend(node.values())
        elif isinstance(node, list):
            stack.extend(node)
    return out


def main() -> None:
    counts: Counter[str] = Counter()
    per_file: list[tuple[str, Counter[str]]] = []
    files = sorted(EVIDENCE.glob("*frames*.json"))
    for path in files:
        try:
            payload = json.loads(path.read_text(encoding="utf-8"))
        except Exception as exc:  # 깨진 파일은 세지 않고 그 사실을 적는다
            print(f"  ⚠️ 읽지 못함: {path.name} — {type(exc).__name__}")
            continue
        local: Counter[str] = Counter()
        for text in agent_finals(payload):
            local[shape(text)] += 1
        if local:
            per_file.append((path.name, local))
            counts.update(local)

    total = sum(counts.values())
    print(f"프레임 파일 {len(files)}개 · 코치 발화 {total}건")
    for name in ("sound", "word", "sentence_only", "none"):
        n = counts[name]
        share = f"{n}/{total}" if total else "0/0"
        print(f"  {name:<14} {share}")
    print("\n파일별(발화가 있는 것만):")
    for name, local in per_file:
        print(f"  {name:<44} {dict(local)}")


main()
